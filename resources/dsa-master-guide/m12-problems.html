def max_depth(root):
    if root is None:
        return 0
    return 1 + max(max_depth(root.left), max_depth(root.right))

def build_tree(vals):
    """Build a binary tree from a level-order list; None means 'no node'."""
    if not vals or vals[0] is None:
        return None
    root = TreeNode(vals[0])
    queue, i = [root], 1
    while queue and i < len(vals):
        node = queue.pop(0)
        if i < len(vals) and vals[i] is not None:
            node.left = TreeNode(vals[i]); queue.append(node.left)
        i += 1
        if i < len(vals) and vals[i] is not None:
            node.right = TreeNode(vals[i]); queue.append(node.right)
        i += 1
    return root

def find_node(root, val):

        return None
    if root.val == val:
        return root
    return find_node(root.left, val) or find_node(root.right, val)

Inorder Traversal

Easy Tree traversal

Return the inorder (left, node, right) traversal of a binary tree.

FieldDetail
DSA conceptTree traversal (Trees & Binary Search Trees)
Input formatRoot.
Output formatList of values.
Constraintsnodes <= 100

Sample input

[2,1,3]

Sample output

[1, 2, 3]

Explanation: Left child 1, root 2, right child 3. For a BST inorder is sorted.

Stuck? Reveal hints one at a time

Hint 1

Preorder = node first; postorder = node last.

Recommended approach (spoiler)

Recurse left, record the node, recurse right.

Python solution (try first)

def inorder(root):
    res = []
    def walk(node):
        if node:
            walk(node.left)
            res.append(node.val)
            walk(node.right)
    walk(root)
    return res

def invert_tree(root):
    if root is None:
        return None
    root.left, root.right = invert_tree(root.right), invert_tree(root.left)
    return root

def level_values(root):
    out, q = [], [root] if root else []
    while q:
        node = q.pop(0)
        out.append(node.val)
        q += [c for c in (node.left, node.right) if c]
    return out

Path Sum

Easy Tree recursion (DFS) LC 112

Return True if some root-to-leaf path adds up to target.

This is a well-known public interview classic (LeetCode-style problem #112). The guide does not claim any company asked it.

FieldDetail
DSA conceptTree recursion (DFS) (Trees & Binary Search Trees)
Input formatRoot and target.
Output formatTrue or False.
Constraintsnodes <= 5000

Sample input

[5,4,8,11,None,13,4,7,2], target 22

Sample output

True

Explanation: Path 5 -> 4 -> 11 -> 2 sums to 22.

Stuck? Reveal hints one at a time

Hint 1

A leaf has no children.

Recommended approach (spoiler)

Subtract the node value from the target and recurse. At a leaf check if what remains is zero.

Python solution (try first)

def has_path_sum(root, target):
    if root is None:
        return False
    if root.left is None and root.right is None:    # leaf
        return target == root.val
    rest = target - root.val
    return has_path_sum(root.left, rest) or has_path_sum(root.right, rest)

Subtree of Another Tree

Easy Tree recursion LC 572 must-do

Return True if some subtree of root has exactly the same structure and values as sub_root.

This is a well-known public interview classic (LeetCode-style problem #572, also on widely shared must-practise lists such as Blind 75). The guide does not claim any company asked it.

Why interviewers ask this: Combines tree comparison with traversal; checks decomposition into helper functions.

FieldDetail
DSA conceptTree recursion (Trees & Binary Search Trees)
Input formatTwo roots.
Output formatTrue or False.
Constraintsnodes <= 2000

Sample input

root [3,4,5,1,2], sub [4,1,2]

Sample output

True

Explanation: The left subtree of the root matches.

Stuck? Reveal hints one at a time

Hint 1

Reuse the "identical trees" helper.

Recommended approach (spoiler)

Write same(a, b). At every node of the big tree check same(node, sub_root).

Python solution (try first)

def is_subtree(root, sub):
    def same(a, b):
        if not a and not b:
            return True
        if not a or not b or a.val != b.val:
            return False
        return same(a.left, b.left) and same(a.right, b.right)
    if not root:
        return sub is None
    return same(root, sub) or is_subtree(root.left, sub) or is_subtree(root.right, sub)

from collections import deque

def level_order(root):
    if not root:
        return []
    res, q = [], deque([root])
    while q:
        level = []
        for _ in range(len(q)):          # exactly this level
            node = q.popleft()
            level.append(node.val)
            if node.left:
                q.append(node.left)
            if node.right:
                q.append(node.right)
        res.append(level)
    return res

Validate a Binary Search Tree

Medium BST property with bounds LC 98 must-do

Return True if the tree is a valid BST (left subtree strictly smaller, right subtree strictly larger, for EVERY ancestor).

This is a well-known public interview classic (LeetCode-style problem #98, also on widely shared must-practise lists such as Blind 75). The guide does not claim any company asked it.

FieldDetail
DSA conceptBST property with bounds (Trees & Binary Search Trees)
Input formatRoot.
Output formatTrue or False.
Constraintsnodes <= 10^4

Sample input

[5,1,4,null,null,3,6]

Sample output

False

Explanation: Node 3 is in the right subtree of 5 but 3 < 5.

Stuck? Reveal hints one at a time

Hint 1

Start with (-inf, +inf).

Hint 2

Equal values are invalid.

Recommended approach (spoiler)

Comparing only parent and child is not enough. Pass down the allowed (low, high) range: going left tightens high, going right tightens low.

Python solution (try first)

def is_valid_bst(root, low=float("-inf"), high=float("inf")):
    if root is None:
        return True
    if not (low < root.val < high):
        return False
    return (is_valid_bst(root.left, low, root.val) and
            is_valid_bst(root.right, root.val, high))

Lowest Common Ancestor

Medium Tree recursion LC 236

Given a binary tree and two values p and q that exist in it, return the value of their lowest common ancestor.

This is a well-known public interview classic (LeetCode-style problem #236). The guide does not claim any company asked it.

FieldDetail
DSA conceptTree recursion (Trees & Binary Search Trees)
Input formatRoot, p, q.
Output formatA value.
Constraintsnodes <= 10^5

Sample input

[3,5,1,6,2,0,8,null,null,7,4], p=5, q=1

Sample output

3

Explanation: They lie in different subtrees of 3.

Stuck? Reveal hints one at a time

Hint 1

What does the function return if a value is found only in the left subtree?

Recommended approach (spoiler)

Recurse: if the current node is p or q return it. If both sides return a node, the current node is the LCA. Otherwise pass up whichever side is non-null.

Python solution (try first)

def lca(root, p, q):
    def go(node):
        if node is None or node.val in (p, q):
            return node
        left, right = go(node.left), go(node.right)
        if left and right:
            return node            # p and q are on different sides
        return left or right
    return go(root).val

def zigzag_level_order(root):
    if not root:
        return []
    res, q, left_to_right = [], deque([root]), True
    while q:
        level = [q.popleft() for _ in range(len(q))]
        vals = [n.val for n in level]

    return res

def serialize_tree(root):
    out = []
    def go(n):
        if not n:
            out.append("#")
            return
        out.append(str(n.val))
        go(n.left)
        go(n.right)
    go(root)

def deserialize_tree(data):
    it = iter(data.split(","))
    def go():
        tok = next(it)
        if tok == "#":
            return None
        node = TreeNode(int(tok))
        node.left = go()
        node.right = go()
        return node
    return go()

    root = TreeNode(vals[0])
    queue, i = [root], 1
    while queue and i < len(vals):
        node = queue.pop(0)
        if i < len(vals) and vals[i] is not None:
            node.left = TreeNode(vals[i]); queue.append(node.left)
        i += 1
        if i < len(vals) and vals[i] is not None:
            node.right = TreeNode(vals[i]); queue.append(node.right)
        i += 1
    return root

def find_node(root, val):
    if not root:
        return None
    if root.val == val:
        return root
    return find_node(root.left, val) or find_node(root.right, val)
