Top-k sampling

  • What: Keep the k highest logits, renormalize with softmax over just those, sample. Temperature 0 means argmax.
  • Key idea: Sort indices by (-logit, index) so ties at the cutoff are deterministic. Sample by walking the cumulative sum until it exceeds a uniform random number.
  • Complexity: O(V log V) with a sort; O(V log k) with a heap (heapq.nlargest).
  • Classic bug: Sampling from the full softmax and then zeroing (forgets to renormalize), or applying softmax before temperature.
  • Check: Logits [0, 1, 2], k = 2: token 0 is dropped, P(token 2) = e^2/(e^2 + e^1) = 0.7311 (tested empirically within 0.02 over 20,000 draws).
"""Top-k sampling."""
import random
from typing import List

from softmax import softmax


def top_k_probs(logits: List[float], k: int, temperature: float = 1.0) -> List[float]:
    """Keep the k largest logits (ties: smaller index), renormalize, zero the rest."""
    if k <= 0:
        raise ValueError("k must be >= 1")
    keep = sorted(range(len(logits)), key=lambda i: (-logits[i], i))[:k]
    sub = softmax([logits[i] for i in keep], temperature)
    probs = [0.0] * len(logits)
    for i, p in zip(keep, sub):
        probs[i] = p
    return probs


def sample_top_k(logits: List[float], k: int, temperature: float = 1.0,
                 rng: random.Random = random) -> int:
    """Draw one token id. temperature == 0 means greedy (argmax)."""
    if temperature == 0:
        return max(range(len(logits)), key=lambda i: (logits[i], -i))
    probs = top_k_probs(logits, k, temperature)
    r, acc = rng.random(), 0.0
    for i, p in enumerate(probs):
        acc += p
        if r < acc:
            return i
    return max(i for i, p in enumerate(probs) if p > 0)   # guard against rounding at acc ~ 1

Top-p (nucleus) sampling

  • What: Softmax (with temperature), sort descending, keep the shortest prefix whose sum reaches p, renormalize, sample.
  • Key idea: Include the token that crosses p, and always keep at least one token.
  • Complexity: O(V log V).
  • Classic bug: Using a strict '> p' check, which drops the crossing token, or forgetting the at-least-one rule for tiny p.
  • Check: p = 0.75 on [0.5, 0.3, 0.15, 0.05] keeps two: [0.625, 0.375, 0, 0].
"""Top-p (nucleus) sampling."""
import random
from typing import List

from softmax import softmax


def top_p_probs(logits: List[float], p: float, temperature: float = 1.0) -> List[float]:
    """Keep the smallest set of most likely tokens whose total probability >= p.

    Sort by probability (ties: smaller index), add tokens until the running
    sum reaches p, always keep at least one, renormalize. Temperature is
    applied BEFORE the cutoff, which is what most inference servers do
    (check your library: the order of processors can differ).
    """
    if not 0 < p <= 1:
        raise ValueError("p must be in (0, 1]")
    probs = softmax(logits, temperature)
    order = sorted(range(len(probs)), key=lambda i: (-probs[i], i))
    keep, acc = [], 0.0
    for i in order:
        keep.append(i)
        acc += probs[i]
        if acc >= p - 1e-12:
            break
    total = sum(probs[i] for i in keep)
    out = [0.0] * len(probs)
    for i in keep:
        out[i] = probs[i] / total
    return out


def sample_top_p(logits: List[float], p: float, temperature: float = 1.0,
                 rng: random.Random = random) -> int:
    probs = top_p_probs(logits, p, temperature)
    r, acc = rng.random(), 0.0
    for i, q in enumerate(probs):
        acc += q
        if r < acc:
            return i
    return max(i for i, q in enumerate(probs) if q > 0)

Numerically stable softmax with temperature

  • What: p_i = exp((z_i - max z)/T) / sum exp((z_j - max z)/T).
  • Key idea: Subtracting the max keeps every exponent at most 0, so exp never overflows and the denominator is at least 1.
  • Complexity: O(V).
  • Classic bug: math.exp(1000) raises OverflowError (numpy returns inf, then nan after division).
  • Check: [2, 1, 0] gives [0.6652, 0.2447, 0.0900]. At T = 0.5: [0.8668, 0.1173, 0.0159]. softmax([1000, 1000]) = [0.5, 0.5].
"""Numerically stable softmax with temperature."""
import math
from typing import List


def softmax(logits: List[float], temperature: float = 1.0) -> List[float]:
    """p_i = exp((z_i - max z) / T) / sum_j exp((z_j - max z) / T).

    Subtracting the max changes nothing mathematically (it cancels in the
    ratio) but keeps every exponent <= 0, so exp never overflows. At least
    one term is exp(0) = 1, so the denominator is never 0.
    """
    if temperature <= 0:
        raise ValueError("temperature must be > 0 (use argmax for T = 0)")
    if not logits:
        return []
    m = max(logits)
    exps = [math.exp((z - m) / temperature) for z in logits]
    total = sum(exps)
    return [e / total for e in exps]

Log-sum-exp

  • What: lse(x) = m + log(sum exp(x_i - m)), with m = max x. log_softmax(z) = z - lse(z).
  • Key idea: The same max-shift trick, in log space. Use log_softmax directly instead of log(softmax(z)), which underflows to log(0) = -inf.
  • Complexity: O(n).
  • Classic bug: Not handling an empty input or all -inf (return -inf).
  • Check: lse([1000, 1000]) = 1000 + ln 2. log_softmax([0, -2000])[1] = -2000 exactly, where log(softmax) would give -inf.
"""Log-sum-exp and log-softmax, computed stably."""
import math
from typing import List


def logsumexp(xs: List[float]) -> float:
    """log(sum(exp(x))) = m + log(sum(exp(x - m))) with m = max(xs).

    Empty input or all -inf returns -inf (the log of an empty sum).
    """
    if not xs:
        return -math.inf
    m = max(xs)
    if m == -math.inf:
        return -math.inf
    return m + math.log(sum(math.exp(x - m) for x in xs))


def log_softmax(logits: List[float]) -> List[float]:
    """log p_i = z_i - logsumexp(z). Use this instead of log(softmax(z)),
    which returns -inf when a probability underflows to 0."""
    lse = logsumexp(logits)
    return [z - lse for z in logits]

Cosine similarity top-k with numpy

  • What: Scores = (M q) / (|rows| |q|). argpartition for the top k, then sort only those k.
  • Key idea: Guard zero vectors with np.divide(..., where=denom > 0). For exact index tie-breaks, widen the candidate set to every row tied with the k-th score before the final lexsort.
  • Complexity: O(n d) for the matrix-vector product, O(n) to partition, O(k log k) to sort.
  • Classic bug: Dividing by zero norms (NaN poisons the ranking), or using argsort on all n when k is small.
  • Check: Rows [1,0], [0,1], [1,1], [0,0], [2,0] with query [1,0]: top 3 = [0, 4, 2] (0 and 4 tie at 1.0, smaller index first).
"""Cosine similarity top-k with numpy."""
import numpy as np


def cosine_top_k(matrix: np.ndarray, query: np.ndarray, k: int):
    """Return (indices, scores) of the k rows most cosine-similar to query.

    cos(a, b) = a.b / (|a| |b|). Normalize once, then one matrix-vector
    product scores every row. Zero vectors get similarity 0 instead of NaN.
    argpartition finds the top k in O(n), then only those k are sorted.
    Ties: smaller index first (lexsort on (index, -score)).
    """
    m = np.asarray(matrix, dtype=np.float64)
    q = np.asarray(query, dtype=np.float64)
    n = m.shape[0]
    if n == 0 or k <= 0:
        return np.array([], dtype=int), np.array([])
    k = min(k, n)
    row_norm = np.linalg.norm(m, axis=1)
    q_norm = np.linalg.norm(q)
    denom = row_norm * q_norm
    scores = np.divide(m @ q, denom, out=np.zeros(n), where=denom > 0)
    if k < n:
        # widen the partition to every row tied with the k-th score so the index tie-break is exact
        kth = np.partition(scores, n - k)[n - k]
        cand = np.nonzero(scores >= kth)[0]
    else:
        cand = np.arange(n)
    order = cand[np.lexsort((cand, -scores[cand]))][:k]
    return order, scores[order]

Reciprocal Rank Fusion

  • What: score(d) = sum over lists of 1/(k + rank), rank 1-based, k = 60.
  • Key idea: Only ranks matter, so no score normalization is needed. A doc missing from a list just gets nothing from it.
  • Complexity: O(total list length + m log m).
  • Classic bug: 0-based ranks, or counting a duplicate id twice within one list.
  • Check: BM25 [d1, d2, d3] and dense [d3, d1, d4] fuse to d1 (1/61 + 1/62), d3 (1/61 + 1/63), d2, d4.
"""Reciprocal Rank Fusion."""
from typing import Dict, Hashable, List, Tuple


def rrf(rankings: List[List[Hashable]], k: int = 60, top_n: int = None) -> List[Tuple[Hashable, float]]:
    """score(d) = sum over lists containing d of 1 / (k + rank(d)), rank 1-based.

    Only ranks are used, never raw scores, so BM25 scores (unbounded) and
    cosine scores (-1..1) can be fused without calibration. k = 60 is the
    value from the original paper (Cormack et al., 2009); larger k flattens
    the advantage of the top positions. A doc missing from a list simply
    gets nothing from that list. If an id repeats inside one list, only its
    first (best) position counts. Ties: id order, so ids must be comparable.
    """
    scores: Dict[Hashable, float] = {}
    for lst in rankings:
        seen = set()
        for rank, doc in enumerate(lst, start=1):
            if doc in seen:
                continue
            seen.add(doc)
            scores[doc] = scores.get(doc, 0.0) + 1.0 / (k + rank)
    out = sorted(scores.items(), key=lambda kv: (-kv[1], kv[0]))
    return out[:top_n] if top_n is not None else out

nDCG@k

  • What: DCG = sum gain_i / log2(i + 1), positions from 1. nDCG = DCG / IDCG.
  • Key idea: IDCG sorts ALL judged relevance values, not just the retrieved ones, so missing a relevant document is penalized.
  • Complexity: O(n log n) for the ideal sort.
  • Classic bug: Using log2(i) with 1-based i (divides by zero at position 1), or building the ideal from the retrieved list only.
  • Check: Ranked [c, b, a] with relevance a = 3, b = 2, c = 1: nDCG@3 = 0.79 (exactly 0.789998).
"""nDCG@k."""
import math
from typing import Dict, Hashable, List


def dcg(gains: List[float]) -> float:
    """sum_i gain_i / log2(i + 1), positions i = 1, 2, 3, ..."""
    return sum(g / math.log2(i + 1) for i, g in enumerate(gains, start=1))


def ndcg_at_k(ranked: List[Hashable], relevance: Dict[Hashable, float], k: int,
              exponential: bool = False) -> float:
    """DCG of the top k divided by the DCG of the ideal top k.

    relevance holds graded judgments for ALL judged docs (unlisted = 0). The
    ideal ordering sorts every judged doc by relevance, not only the ones the
    system retrieved; otherwise a system that misses relevant docs gets
    flattered. exponential=True uses gain 2^rel - 1 (common in web search),
    the default uses gain = rel. Returns 0.0 when nothing is relevant.
    """
    gain = (lambda r: 2 ** r - 1) if exponential else (lambda r: r)
    got = [gain(relevance.get(d, 0)) for d in ranked[:k]]
    ideal = sorted((gain(r) for r in relevance.values()), reverse=True)[:k]
    idcg = dcg(ideal)
    return dcg(got) / idcg if idcg > 0 else 0.0

Mean Reciprocal Rank

  • What: RR = 1/(position of the first relevant result), 0 if none. MRR = mean over queries.
  • Key idea: Only the first relevant hit matters. Queries with no hit count as 0 (do not drop them).
  • Complexity: O(total results).
  • Classic bug: Averaging only over queries with a hit, which inflates MRR.
  • Check: First hits at ranks 1, 2 and none give (1 + 0.5 + 0)/3 = 0.5.
"""Mean Reciprocal Rank."""
from typing import Hashable, List, Set


def reciprocal_rank(ranked: List[Hashable], relevant: Set[Hashable], k: int = None) -> float:
    """1 / (1-based position of the FIRST relevant result), 0 if none in the top k."""
    for pos, d in enumerate(ranked[:k] if k else ranked, start=1):
        if d in relevant:
            return 1.0 / pos
    return 0.0


def mean_reciprocal_rank(runs: List[List[Hashable]], qrels: List[Set[Hashable]], k: int = None) -> float:
    """Average of reciprocal_rank over queries. Queries with no hit count as 0."""
    if not runs:
        return 0.0
    return sum(reciprocal_rank(r, q, k) for r, q in zip(runs, qrels)) / len(runs)

MMR diversification

  • What: Pick argmax of lam x sim(q, d) - (1 - lam) x max similarity to anything already selected.
  • Key idea: Keep a running max-similarity array, updated once per selection, so each round is O(n) instead of O(n x |S|).
  • Complexity: O(k x n).
  • Classic bug: Using the sum instead of the max similarity to selected docs, or forgetting that the first pick is purely by relevance.
  • Check: Two near-duplicates (similarity 0.99) and a different doc: lam = 1 picks both duplicates, lam = 0.5 picks [0, 2].
"""Maximal Marginal Relevance diversification."""
from typing import List


def mmr(query_sim: List[float], doc_sim: List[List[float]], k: int, lam: float = 0.7) -> List[int]:
    """Greedy selection: argmax_d [ lam * sim(q, d) - (1 - lam) * max_{s in S} sim(d, s) ].

    lam = 1 is plain relevance ranking, lam = 0 is pure diversity. The first
    pick is always the most relevant doc (S is empty, penalty 0). Ties:
    smaller index. Returns picks in selection order. O(k * n).
    """
    n = len(query_sim)
    selected: List[int] = []
    max_sim = [float("-inf")] * n  # max similarity to anything selected so far
    remaining = set(range(n))
    while remaining and len(selected) < k:
        def gain(d):
            penalty = max_sim[d] if selected else 0.0
            return (lam * query_sim[d] - (1 - lam) * penalty, -d)
        best = max(remaining, key=gain)
        selected.append(best)
        remaining.discard(best)
        for d in remaining:
            max_sim[d] = max(max_sim[d], doc_sim[d][best])
    return selected

Token bucket rate limiter

  • What: Capacity C tokens, refilled at rate r per second. A request spends cost tokens if available.
  • Key idea: Refill lazily from elapsed time (tokens = min(C, tokens + dt x r)). There is no background thread. Inject the clock for tests.
  • Complexity: O(1) per call.
  • Classic bug: Refilling past capacity after idle periods, or using wall-clock time (time.time) instead of a monotonic clock.
  • Check: rate 2/s, capacity 5: a burst of 5 passes, the 6th fails. After 0.5 s exactly one more passes.
"""Token bucket rate limiter."""
import time
from typing import Callable, Optional


class TokenBucket:
    """Holds up to `capacity` tokens and refills at `rate` tokens per second.

    allow(cost) spends `cost` tokens if available. Bursts up to capacity
    are allowed; the long-run average is limited to `rate`. Refill is
    computed lazily from elapsed time, so there is no background thread.
    For LLM APIs, cost can be the request's token count (tokens-per-minute
    limits) instead of 1 (requests-per-minute limits).
    """

    def __init__(self, rate: float, capacity: float, clock: Callable[[], float] = time.monotonic):
        self.rate, self.capacity, self.clock = rate, capacity, clock
        self.tokens = capacity
        self.last = clock()

    def _refill(self, now: float) -> None:
        if now > self.last:
            self.tokens = min(self.capacity, self.tokens + (now - self.last) * self.rate)
            self.last = now

    def allow(self, cost: float = 1.0, now: Optional[float] = None) -> bool:
        self._refill(self.clock() if now is None else now)
        if cost <= self.tokens:
            self.tokens -= cost
            return True
        return False

    def wait_time(self, cost: float = 1.0, now: Optional[float] = None) -> float:
        """Seconds until `cost` tokens will be available (0 if available now)."""
        self._refill(self.clock() if now is None else now)
        if cost > self.capacity:
            return float("inf")
        return max(0.0, (cost - self.tokens) / self.rate)

LRU cache (and a block prefix cache)

  • What: OrderedDict: move_to_end on get and put, popitem(last=False) to evict the least recently used.
  • Key idea: Prompt caches key each block by the ENTIRE prefix up to it, so a block is reused only when everything before it matches. That is why only shared prefixes hit.
  • Complexity: O(1) get and put.
  • Classic bug: Not refreshing recency on get, or evicting before inserting an existing key.
  • Check: Capacity 2: put a, put b, get a, put c evicts b. The prefix cache reuses 12 tokens of a shared 12-token system prompt, and 0 if the first token changes.
"""LRU cache, and a block-level prefix cache built on it (the idea behind prompt/prefix caching)."""
from collections import OrderedDict
from typing import Hashable, List, Optional, Sequence


class LRUCache:
    """O(1) get/put. OrderedDict keeps keys in recency order; the front is least recent."""

    def __init__(self, capacity: int):
        if capacity <= 0:
            raise ValueError("capacity must be >= 1")
        self.capacity = capacity
        self.data: "OrderedDict[Hashable, object]" = OrderedDict()

    def get(self, key: Hashable) -> Optional[object]:
        if key not in self.data:
            return None
        self.data.move_to_end(key)          # mark as most recently used
        return self.data[key]

    def put(self, key: Hashable, value: object) -> Optional[Hashable]:
        """Insert or update. Returns the evicted key, if any."""
        if key in self.data:
            self.data.move_to_end(key)
            self.data[key] = value
            return None
        self.data[key] = value
        if len(self.data) > self.capacity:
            evicted, _ = self.data.popitem(last=False)
            return evicted
        return None


class PrefixCache:
    """Caches prompt prefixes in fixed-size token blocks, keyed by the whole prefix.

    The key for block j is the tuple of ALL tokens up to the end of block j,
    so a block is only reused when everything before it matches too. That
    is why prompt caching only helps with a shared PREFIX: change one early
    token and every later block key changes. Real servers hash the prefix
    instead of storing the tuple (vLLM-style automatic prefix caching works
    on this block principle; details vary by server and version).
    """

    def __init__(self, block: int, capacity_blocks: int):
        self.block = block
        self.lru = LRUCache(capacity_blocks)

    def _keys(self, tokens: Sequence[int]) -> List[tuple]:
        full = len(tokens) // self.block
        return [tuple(tokens[:(j + 1) * self.block]) for j in range(full)]

    def lookup(self, tokens: Sequence[int]) -> int:
        """Number of leading tokens whose KV could be reused (block aligned)."""
        hit = 0
        for key in self._keys(tokens):
            if self.lru.get(key) is None:
                break
            hit += self.block
        return hit

    def insert(self, tokens: Sequence[int]) -> None:
        for key in self._keys(tokens):
            self.lru.put(key, True)

One BPE merge step

  • What: Count adjacent symbol pairs weighted by word frequency, take the most frequent, replace it everywhere left to right.
  • Key idea: Merge non-overlapping occurrences scanning left to right (aaa with pair aa gives aa, a). Ties need a rule: here, the lexicographically smallest pair.
  • Complexity: O(total symbols) per step.
  • Classic bug: Overlapping merges, or forgetting to weight pair counts by word frequency.
  • Check: {low: 5, lower: 2, newest: 6, widest: 3}: es and st both count 9, so the tie gives (e, s). The next step merges (es, t).
"""One BPE training merge step."""
from collections import Counter
from typing import Dict, Optional, Tuple

Word = Tuple[str, ...]


def most_frequent_pair(vocab: Dict[Word, int]) -> Optional[Tuple[str, str]]:
    """Count adjacent symbol pairs weighted by word frequency.
    Ties: lexicographically smallest pair (real tokenizers pick their own rule)."""
    pairs: Counter = Counter()
    for word, freq in vocab.items():
        for a, b in zip(word, word[1:]):
            pairs[(a, b)] += freq
    if not pairs:
        return None
    best = max(pairs.values())
    return min(p for p, c in pairs.items() if c == best)


def merge_pair(vocab: Dict[Word, int], pair: Tuple[str, str]) -> Dict[Word, int]:
    """Replace every non-overlapping occurrence of pair, scanning left to right."""
    a, b = pair
    out: Dict[Word, int] = {}
    for word, freq in vocab.items():
        merged, i = [], 0
        while i < len(word):
            if i + 1 < len(word) and word[i] == a and word[i + 1] == b:
                merged.append(a + b)
                i += 2
            else:
                merged.append(word[i])
                i += 1
        key = tuple(merged)
        out[key] = out.get(key, 0) + freq
    return out


def bpe_step(vocab: Dict[Word, int]):
    """One training step: find the best pair and merge it. Returns (pair, new_vocab)."""
    pair = most_frequent_pair(vocab)
    if pair is None:
        return None, vocab
    return pair, merge_pair(vocab, pair)

KV cache size calculator

  • What: 2 x layers x kv_heads x head_dim x seq_len x batch x bytes.
  • Key idea: Use KV heads (GQA), not query heads. Keep GiB (2^30) and GB (10^9) separate.
  • Complexity: O(1).
  • Classic bug: Query heads instead of KV heads, or forgetting the 2.
  • Check: 32 layers, 8 KV heads, dim 128, fp16, batch 4 x 8192 gives exactly 4 GiB.
"""KV cache and weight memory calculators."""

DTYPE_BYTES = {"fp32": 4, "fp16": 2, "bf16": 2, "fp8": 1, "int8": 1, "int4": 0.5}
GiB = 1024 ** 3
GB = 1000 ** 3


def kv_cache_bytes(layers: int, kv_heads: int, head_dim: int, seq_len: int,
                   batch: int, bytes_per_elem: float) -> float:
    """2 (K and V) x layers x kv_heads x head_dim x seq_len x batch x bytes.

    kv_heads is the number of KEY/VALUE heads: equal to the query heads for
    MHA, 1 for MQA, and a small group count for GQA (for example 8).
    """
    return 2 * layers * kv_heads * head_dim * seq_len * batch * bytes_per_elem


def kv_bytes_per_token(layers: int, kv_heads: int, head_dim: int, bytes_per_elem: float) -> float:
    return 2 * layers * kv_heads * head_dim * bytes_per_elem


def weight_bytes(params: float, dtype: str) -> float:
    """Weights only. Real int4 formats also store scales/zero-points (a few % more)."""
    return params * DTYPE_BYTES[dtype]


def max_tokens_in_cache(free_bytes: float, layers: int, kv_heads: int, head_dim: int,
                        bytes_per_elem: float) -> int:
    """How many tokens (summed over all sequences in the batch) fit in free memory."""
    return int(free_bytes // kv_bytes_per_token(layers, kv_heads, head_dim, bytes_per_elem))

Exponential backoff with jitter

  • What: delay = uniform(0, min(cap, base x 2^attempt)). Retry only transient errors. Bounded attempts.
  • Key idea: Jitter de-synchronizes clients. Side-effecting calls need an idempotency key reused on every attempt (part 05).
  • Complexity: O(attempts).
  • Classic bug: Retrying non-retryable errors (validation), unbounded retries, or no jitter.
  • Check: A function failing twice with TimeoutError, then succeeding: 3 calls, 2 sleeps. A ValueError propagates immediately.
"""Retry with exponential backoff and full jitter."""
import random
import time
from typing import Callable, Tuple, Type


def backoff_delay(attempt: int, base: float = 0.5, cap: float = 30.0,
                  rng: random.Random = random) -> float:
    """Full jitter: uniform(0, min(cap, base * 2^attempt)), attempt = 0, 1, 2...

    Exponential growth gives a struggling server room. Jitter spreads
    clients out so they do not all retry at the same instant (the
    "thundering herd"). The cap keeps one delay from growing without bound.
    """
    return rng.uniform(0, min(cap, base * (2 ** attempt)))


def retry(fn: Callable[[], object], max_attempts: int = 5,
          retry_on: Tuple[Type[BaseException], ...] = (TimeoutError, ConnectionError),
          base: float = 0.5, cap: float = 30.0,
          sleep: Callable[[float], None] = time.sleep, rng: random.Random = random):
    """Call fn until it succeeds, retrying only on retry_on errors.

    Only safe for idempotent operations: if fn has side effects (a payment,
    a refund), pass the same idempotency key on every attempt so a retry
    after a timeout cannot apply the effect twice.
    """
    for attempt in range(max_attempts):
        try:
            return fn()
        except retry_on:
            if attempt == max_attempts - 1:
                raise
            sleep(backoff_delay(attempt, base, cap, rng))

Text chunker with overlap

  • What: Windows of size tokens, step = size - overlap. Stop once a window reaches the end.
  • Key idea: Count: 1 + ceil((n - size)/step) for n > size. Checking 'did this window reach the end' prevents a tiny duplicate tail chunk.
  • Complexity: O(n) windows (O(n x size / step) copying).
  • Classic bug: Allowing overlap >= size (infinite loop), or emitting a final chunk that is fully contained in the previous one.
  • Check: 10 tokens, size 4, overlap 1: [0-3], [3-6], [6-9]. Verified against the count formula for all n < 40, sizes up to 8.
"""Fixed-size text chunker with overlap."""
from typing import List, Sequence


def chunk_tokens(tokens: Sequence, size: int, overlap: int) -> List[Sequence]:
    """Windows of `size` tokens; consecutive windows share `overlap` tokens.

    step = size - overlap. Stops once a window reaches the end, so the tail
    is covered exactly once and no tiny duplicate chunk is produced.
    Number of chunks for n > size: 1 + ceil((n - size) / step).
    """
    if size <= 0 or not 0 <= overlap < size:
        raise ValueError("need size > 0 and 0 <= overlap < size")
    step = size - overlap
    out = []
    start = 0
    while True:
        out.append(tokens[start:start + size])
        if start + size >= len(tokens):
            break
        start += step
    return out if len(tokens) else []


def chunk_words(text: str, size: int, overlap: int) -> List[str]:
    """Whitespace-word version (real pipelines count model tokens, and prefer
    splitting on sentence or heading boundaries)."""
    return [" ".join(c) for c in chunk_tokens(text.split(), size, overlap)]

Streaming SSE parser

  • What: Lines end with \n, \r\n or \r. 'field: value'. data lines are joined with \n. A blank line dispatches. A line starting with ':' is a comment.
  • Key idea: Buffer partial lines across chunks, and remember a trailing \r so a \n arriving at the start of the next chunk is not treated as a second (blank) line.
  • Complexity: O(n) overall (a regex finds each line end).
  • Classic bug: Splitting only on \n, dispatching on every data line, or losing a partial line at a chunk boundary.
  • Check: Tested on every possible split point of a stream mixing \n, \r\n and \r, and when fed one character at a time.
"""Incremental Server-Sent Events parser (the wire format LLM APIs stream over)."""
import re
from typing import Dict, List, Optional

_EOL = re.compile(r"\r\n|\r|\n")


class SSEParser:
    """feed(chunk) -> list of complete events {"event", "data", "id"}.

    Rules (from the HTML Living Standard's EventSource section):
    - Lines end with \\n, \\r\\n or \\r. A chunk may end mid-line, or between \\r and \\n.
    - A blank line dispatches the event if any data was collected.
    - "field: value": one leading space after the colon is removed.
    - "data" lines are joined with \\n. A line starting with ":" is a comment.
    - Event type defaults to "message". "id" persists across events.
    """

    def __init__(self) -> None:
        self.buf = ""
        self.data: List[str] = []
        self.event = ""
        self.last_id = ""
        self.retry: Optional[int] = None
        self._skip_lf = False

    def feed(self, chunk: str) -> List[Dict[str, str]]:
        out: List[Dict[str, str]] = []
        if chunk and self._skip_lf:
            if chunk.startswith("\n"):
                chunk = chunk[1:]
            self._skip_lf = False
        self.buf += chunk
        while True:
            m = _EOL.search(self.buf)
            if not m:
                break
            line = self.buf[:m.start()]
            if m.group() == "\r" and m.end() == len(self.buf):
                self._skip_lf = True            # a "\n" at the start of the next chunk belongs to this "\r"
            self.buf = self.buf[m.end():]
            ev = self._line(line)
            if ev:
                out.append(ev)
        return out

    def _line(self, line: str) -> Optional[Dict[str, str]]:
        if line == "":
            if not self.data:
                self.event = ""
                return None
            ev = {"event": self.event or "message", "data": "\n".join(self.data), "id": self.last_id}
            self.data, self.event = [], ""
            return ev
        if line.startswith(":"):
            return None                        # comment / keep-alive
        field, sep, value = line.partition(":")
        if sep and value.startswith(" "):
            value = value[1:]
        if field == "data":
            self.data.append(value)
        elif field == "event":
            self.event = value
        elif field == "id" and "\0" not in value:
            self.last_id = value
        elif field == "retry" and value.isdigit():
            self.retry = int(value)
        return None

Sliding-window token budget trimmer

  • What: Always keep system messages. Then keep the newest messages, walking backward, while they fit. Stop at the first that does not.
  • Key idea: Stop rather than skip, so the kept history is a contiguous recent window. Raise if the newest message cannot fit: silently dropping the user's question is the worst failure.
  • Complexity: O(n) messages (plus token counting).
  • Classic bug: Dropping the system prompt first, or keeping an older message after skipping a newer one (a gap in the conversation).
  • Check: Budget 8 with sizes system 2, user 3, assistant 2, user 4 keeps system, assistant, last user.
"""Sliding-window token budget trimmer for chat history."""
from typing import Callable, Dict, List


def trim_history(messages: List[Dict[str, str]], budget: int,
                 count_tokens: Callable[[str], int]) -> List[Dict[str, str]]:
    """Keep system messages and the newest messages that fit; drop the oldest.

    - System messages are always kept (they carry the instructions).
    - Walk the rest from newest to oldest and keep each whole message while it
      fits; stop at the first one that does not (never skip a gap, which would
      break the conversation's order of events).
    - The newest message must fit, otherwise raise: silently dropping the
      user's actual question is the worst possible failure.
    Returns kept messages in their original order.
    """
    system = [i for i, m in enumerate(messages) if m["role"] == "system"]
    used = sum(count_tokens(messages[i]["content"]) for i in system)
    if used > budget:
        raise ValueError("system messages alone exceed the budget")
    keep = set(system)
    others = [i for i in range(len(messages)) if i not in keep]
    for n, i in enumerate(reversed(others)):
        cost = count_tokens(messages[i]["content"])
        if used + cost > budget:
            if n == 0:
                raise ValueError("newest message does not fit in the budget")
            break
        used += cost
        keep.add(i)
    return [messages[i] for i in sorted(keep)]

Scaled dot-product attention (with causal mask)

  • What: out = softmax(Q K^T / sqrt(d_k) + mask) V, one row per query.
  • Key idea: Divide by sqrt(d_k) so dot products do not grow with dimension and saturate the softmax. The causal mask sets future positions to -inf, so their weight is exactly 0.
  • Complexity: O(n_q x n_k x d) time, O(n_q x n_k) for the score matrix (FlashAttention avoids materializing it).
  • Classic bug: Forgetting the scale, applying the mask after the softmax (the weights then no longer sum to 1), or masking j >= i instead of j > i (which hides the token from itself).
  • Check: q = [1, 0] over keys [1,0], [0,1], [1,1] gives weights [0.4011, 0.1978, 0.4011] and output [0.8022, 0.5989]. Matches a numpy reference on 20 random causal cases.
"""Scaled dot-product attention (single head) with an optional causal mask, pure Python."""
import math
from typing import List

Matrix = List[List[float]]


def attention(Q: Matrix, K: Matrix, V: Matrix, causal: bool = False) -> Matrix:
    """out = softmax(Q K^T / sqrt(d_k) + mask) V

    Q: n_q x d_k, K: n_k x d_k, V: n_k x d_v. With causal=True, query i may
    only attend to keys 0..i (requires n_q == n_k), which is how a decoder
    stops token i from seeing the future. Each softmax row is max-shifted for
    stability. Cost: O(n_q x n_k x d), the quadratic term in sequence length.
    """
    d_k = len(K[0])
    scale = 1.0 / math.sqrt(d_k)
    out = []
    for i, q in enumerate(Q):
        scores = []
        for j, k in enumerate(K):
            if causal and j > i:
                scores.append(-math.inf)           # masked: weight becomes exactly 0
            else:
                scores.append(sum(a * b for a, b in zip(q, k)) * scale)
        m = max(scores)
        exps = [math.exp(s - m) if s != -math.inf else 0.0 for s in scores]
        total = sum(exps)
        weights = [e / total for e in exps]
        out.append([sum(w * v[c] for w, v in zip(weights, V)) for c in range(len(V[0]))])
    return out

Transformer parameter count

  • What: Per layer: attention 2d^2 + 2d x kv_dim, SwiGLU MLP 3 x d x ffn, two norms. Plus embeddings, LM head, final norm.
  • Key idea: GQA shrinks only the K and V projections. Untied embeddings count the vocab x d matrix twice.
  • Complexity: O(1).
  • Classic bug: Using 2 x d x ffn for a SwiGLU MLP (it has three matrices), or forgetting the LM head.
  • Check: Llama 2 7B = 6,738,415,616 and Llama 3 8B = 8,030,261,248, exactly the published counts.
"""Parameter count of a Llama-style decoder (no biases, RMSNorm, SwiGLU MLP)."""


def llama_params(vocab: int, d: int, layers: int, ffn: int, n_heads: int, n_kv_heads: int,
                 tied_embeddings: bool = False) -> int:
    """Count every weight:

    per layer:
      attention  q: d x d, o: d x d, k and v: d x (n_kv_heads x head_dim) each
      MLP (SwiGLU) gate, up: d x ffn, down: ffn x d  -> 3 x d x ffn
      2 RMSNorm weight vectors of size d
    plus the token embedding (vocab x d), the LM head (vocab x d unless tied)
    and the final norm (d).
    Rule of thumb for MHA with ffn = 4d: about 12 x layers x d^2 + embeddings.
    """
    head_dim = d // n_heads
    kv_dim = n_kv_heads * head_dim
    attn = 2 * d * d + 2 * d * kv_dim
    mlp = 3 * d * ffn
    per_layer = attn + mlp + 2 * d
    embed = vocab * d * (1 if tied_embeddings else 2)
    return layers * per_layer + embed + d

Precision, recall, F1 and ROC-AUC

  • What: Confusion counts, then P = TP/(TP+FP), R = TP/(TP+FN), F1 = 2PR/(P+R). AUC = P(random positive scores above random negative).
  • Key idea: AUC via ranks (Mann-Whitney U) is O(n log n). Ties get their average rank, which counts tied pairs as 1/2.
  • Complexity: O(n) for the counts, O(n log n) for AUC.
  • Classic bug: Dividing by zero when nothing is predicted positive, or the O(n_pos x n_neg) pair loop on large data.
  • Check: TP 40, FP 10, FN 20, TN 930: P 0.8, R 0.667, F1 0.7273 (accuracy 0.97 looks great but misleads). The rank AUC matches brute-force pairs on 300 random cases with ties.
"""Confusion-matrix metrics and ROC-AUC from scratch."""
from typing import Dict, List, Sequence


def confusion(y_true: Sequence[int], y_pred: Sequence[int]) -> Dict[str, int]:
    tp = sum(1 for t, p in zip(y_true, y_pred) if t == 1 and p == 1)
    fp = sum(1 for t, p in zip(y_true, y_pred) if t == 0 and p == 1)
    fn = sum(1 for t, p in zip(y_true, y_pred) if t == 1 and p == 0)
    tn = sum(1 for t, p in zip(y_true, y_pred) if t == 0 and p == 0)
    return {"tp": tp, "fp": fp, "fn": fn, "tn": tn}


def precision_recall_f1(y_true: Sequence[int], y_pred: Sequence[int], beta: float = 1.0):
    """precision = tp/(tp+fp), recall = tp/(tp+fn), F_beta = (1+b^2)PR/(b^2 P + R).
    Undefined ratios (0/0) are returned as 0.0, the common library convention."""
    c = confusion(y_true, y_pred)
    p = c["tp"] / (c["tp"] + c["fp"]) if c["tp"] + c["fp"] else 0.0
    r = c["tp"] / (c["tp"] + c["fn"]) if c["tp"] + c["fn"] else 0.0
    b2 = beta * beta
    f = (1 + b2) * p * r / (b2 * p + r) if (b2 * p + r) else 0.0
    return p, r, f


def roc_auc(y_true: Sequence[int], scores: Sequence[float]) -> float:
    """AUC = P(score of a random positive > score of a random negative), ties count 1/2.

    Computed with ranks (Mann-Whitney U) in O(n log n): sort by score, give
    tied scores their average rank, then
        AUC = (sum of positive ranks - n_pos (n_pos + 1) / 2) / (n_pos x n_neg)
    """
    n = len(scores)
    order = sorted(range(n), key=lambda i: scores[i])
    ranks: List[float] = [0.0] * n
    i = 0
    while i < n:
        j = i
        while j + 1 < n and scores[order[j + 1]] == scores[order[i]]:
            j += 1
        avg = (i + j) / 2 + 1                     # 1-based average rank of the tie group
        for k in range(i, j + 1):
            ranks[order[k]] = avg
        i = j + 1
    n_pos = sum(1 for t in y_true if t == 1)
    n_neg = n - n_pos
    if n_pos == 0 or n_neg == 0:
        raise ValueError("AUC needs both classes")
    s = sum(r for r, t in zip(ranks, y_true) if t == 1)
    return (s - n_pos * (n_pos + 1) / 2) / (n_pos * n_neg)

Logistic regression with gradient descent

  • What: p = sigmoid(w.x + b), minimize mean binary cross-entropy. Gradient: mean((p - y) x), plus l2 x w if regularized.
  • Key idea: The gradient has the same shape as linear regression's. Use a numerically stable sigmoid, never exp of a large positive number.
  • Complexity: O(epochs x n x d).
  • Classic bug: Regularizing the bias, using MSE loss (non-convex with sigmoid), or a learning rate so large the loss oscillates.
  • Check: Separable toy data is classified perfectly, and the analytic gradient matches a finite-difference check to 1e-6.
"""Logistic regression trained with batch gradient descent, pure Python."""
import math
from typing import List, Sequence, Tuple


def sigmoid(z: float) -> float:
    # stable for large |z|: never calls exp on a big positive number
    if z >= 0:
        return 1.0 / (1.0 + math.exp(-z))
    e = math.exp(z)
    return e / (1.0 + e)


def log_loss(X: Sequence[Sequence[float]], y: Sequence[int], w: List[float], b: float, l2: float = 0.0) -> float:
    eps = 1e-12
    total = 0.0
    for xi, yi in zip(X, y):
        p = sigmoid(sum(a * c for a, c in zip(w, xi)) + b)
        total += -(yi * math.log(p + eps) + (1 - yi) * math.log(1 - p + eps))
    return total / len(y) + 0.5 * l2 * sum(c * c for c in w)


def fit(X: Sequence[Sequence[float]], y: Sequence[int], lr: float = 0.1, epochs: int = 500,
        l2: float = 0.0) -> Tuple[List[float], float]:
    """Minimize mean binary cross-entropy (+ optional L2 on w, not on b).

    The gradient is remarkably simple: dL/dw = mean((p - y) x) + l2 w and
    dL/db = mean(p - y). Same form as linear regression's, with p = sigmoid(w.x + b).
    """
    n, d = len(X), len(X[0])
    w, b = [0.0] * d, 0.0
    for _ in range(epochs):
        gw, gb = [0.0] * d, 0.0
        for xi, yi in zip(X, y):
            err = sigmoid(sum(a * c for a, c in zip(w, xi)) + b) - yi
            for j in range(d):
                gw[j] += err * xi[j]
            gb += err
        w = [w[j] - lr * (gw[j] / n + l2 * w[j]) for j in range(d)]
        b -= lr * gb / n
    return w, b


def predict_proba(X: Sequence[Sequence[float]], w: List[float], b: float) -> List[float]:
    return [sigmoid(sum(a * c for a, c in zip(w, xi)) + b) for xi in X]

k-means with k-means++ init

  • What: Repeat: assign points to the nearest centroid, then move each centroid to the mean of its points. Stop when assignments stop changing.
  • Key idea: Each step lowers the within-cluster sum of squares, so it converges, but only to a local optimum. k-means++ spreads the initial centroids out to avoid bad starts.
  • Complexity: O(iters x n x k x d).
  • Classic bug: Not handling empty clusters, forgetting to scale features (distance is dominated by large-unit features), or assuming the result is the global optimum.
  • Check: Three well-separated blobs of 30 points are each assigned to exactly one cluster.
"""k-means (Lloyd's algorithm) with k-means++ initialization, pure Python."""
import random
from typing import List, Sequence, Tuple

Point = Sequence[float]


def _d2(a: Point, b: Point) -> float:
    return sum((x - y) ** 2 for x, y in zip(a, b))


def kmeans(points: List[Point], k: int, iters: int = 100, seed: int = 0) -> Tuple[List[List[float]], List[int]]:
    """Alternate two steps until assignments stop changing:
    1. assign each point to its nearest centroid (squared Euclidean),
    2. move each centroid to the mean of its points.
    Each step can only lower the within-cluster sum of squares, so it
    converges, but to a LOCAL optimum that depends on the initialization.
    k-means++ picks initial centroids far apart (probability proportional to
    squared distance), which makes bad local optima much less likely.
    """
    rng = random.Random(seed)
    centroids = [list(rng.choice(points))]
    while len(centroids) < k:
        d2 = [min(_d2(p, c) for c in centroids) for p in points]
        total = sum(d2)
        r, acc = rng.random() * total, 0.0
        for p, w in zip(points, d2):
            acc += w
            if acc >= r:
                centroids.append(list(p))
                break
    assign: List[int] = [-1] * len(points)
    for _ in range(iters):
        new = [min(range(k), key=lambda c: (_d2(p, centroids[c]), c)) for p in points]
        if new == assign:
            break
        assign = new
        for c in range(k):
            members = [p for p, a in zip(points, assign) if a == c]
            if members:                               # keep an empty cluster's centroid in place
                centroids[c] = [sum(col) / len(members) for col in zip(*members)]
    return centroids, assign


def inertia(points: List[Point], centroids: List[List[float]], assign: List[int]) -> float:
    """Within-cluster sum of squared distances, the objective k-means minimizes."""
    return sum(_d2(p, centroids[a]) for p, a in zip(points, assign))
