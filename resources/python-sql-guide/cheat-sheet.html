SQL

Query skeleton and evaluation order

SELECT   cols, aggregates, window functions   -- 5
FROM     table JOIN table ON ...              -- 1
WHERE    row filter                           -- 2
GROUP BY cols                                 -- 3
HAVING   group filter                         -- 4
ORDER BY cols                                 -- 6
LIMIT n OFFSET m;                             -- 7

Joins and sets

WantWrite
Only matching rowsINNER JOIN
All left rowsLEFT JOIN (filter right side in ON)
Rows with no matchLEFT JOIN ... WHERE r.id IS NULL or NOT EXISTS
All combinationsCROSS JOIN
Combine resultsUNION ALL (keep dups), UNION, INTERSECT, EXCEPT

Patterns

-- top N per group
SELECT * FROM (SELECT *, DENSE_RANK() OVER (PARTITION BY g ORDER BY v DESC) r FROM t) WHERE r <= 3;
-- running total, previous row
SUM(x) OVER (ORDER BY d), LAG(x) OVER (ORDER BY d)
-- conditional count
SUM(CASE WHEN status = 'paid' THEN 1 ELSE 0 END)
-- gaps and islands
d - ROW_NUMBER() OVER (PARTITION BY u ORDER BY d)
-- upsert
INSERT ... ON CONFLICT (k) DO UPDATE SET v = excluded.v
-- keyset pagination
WHERE id > :last ORDER BY id LIMIT 20

NULL rules

x = NULL is never true: use IS NULL. COUNT(col) skips NULLs. NOT IN with a NULL returns nothing: use NOT EXISTS. COALESCE(a, b) first non-null.
