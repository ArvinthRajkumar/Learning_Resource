1. Word frequency: top k

Return the k most common words, ignoring case and punctuation; break ties alphabetically.

import re
from collections import Counter

def top_words(text, k):
    words = re.findall(r"[a-z']+", text.lower())
    counts = Counter(words)
    return sorted(counts.items(), key=lambda kv: (-kv[1], kv[0]))[:k]

print(top_words("The cat and the hat. The bat sat; the cat ran!", 3))

Output

[('the', 4), ('cat', 2), ('and', 1)]

Time O(n log n) for the sort (use heapq.nsmallest with the same key for O(n log k)).

2. Group anagrams

from collections import defaultdict
def group_anagrams(words):
    groups = defaultdict(list)
    for w in words:
        groups["".join(sorted(w))].append(w)
    return sorted(groups.values())
print(group_anagrams(["eat", "tea", "tan", "ate", "nat", "bat"]))

Output

[['bat'], ['eat', 'tea', 'ate'], ['tan', 'nat']]

3. Flatten an arbitrarily nested list (lazily)

def flatten(items):
    for x in items:
        if isinstance(x, (list, tuple)):
            yield from flatten(x)
        else:
            yield x
print(list(flatten([1, [2, (3, [4])], 5, [[6]]])))

Output

[1, 2, 3, 4, 5, 6]

4. Chunk an iterable and slide a window

from itertools import islice
from collections import deque

def chunked(iterable, n):
    it = iter(iterable)
    while chunk := list(islice(it, n)):
        yield chunk

def windows(iterable, n):
    it = iter(iterable)
    win = deque(islice(it, n), maxlen=n)
    if len(win) == n:
        yield tuple(win)
    for x in it:
        win.append(x)
        yield tuple(win)

print(list(chunked(range(7), 3)))
print(list(windows([1, 2, 3, 4, 5], 3)))

Output

[[0, 1, 2], [3, 4, 5], [6]]
[(1, 2, 3), (2, 3, 4), (3, 4, 5)]

Both work on any iterable, including generators of unknown length, in constant memory.

5. Run-length encoding

from itertools import groupby
def rle(s):
    return "".join(f"{len(list(g))}{ch}" for ch, g in groupby(s))
def unrle(s):
    import re
    return "".join(ch * int(n) for n, ch in re.findall(r"(\d+)(\D)", s))
enc = rle("aaabccdddd")
print(enc, unrle(enc) == "aaabccdddd")

7. Deep merge of dictionaries

def deep_merge(a, b):
    out = dict(a)
    for k, v in b.items():
        if k in out and isinstance(out[k], dict) and isinstance(v, dict):
            out[k] = deep_merge(out[k], v)
        else:
            out[k] = v
    return out
print(deep_merge({"a": 1, "b": {"x": 1, "y": 2}}, {"b": {"y": 3, "z": 4}, "c": 5}))

Output

{'a': 1, 'b': {'x': 1, 'y': 3, 'z': 4}, 'c': 5}

It returns a new dict and does not mutate the inputs.

8. A memoising decorator with expiry

import functools

def cache_with_ttl(ttl, clock):
    def deco(fn):
        store = {}
        @functools.wraps(fn)
        def wrapper(*args):
            now = clock()
            if args in store and now - store[args][1] < ttl:
                return store[args][0]
            value = fn(*args)
            store[args] = (value, now)
            return value
        return wrapper
    return deco

t = [0]
calls = []
@cache_with_ttl(10, lambda: t[0])
def slow(x):
    calls.append(x)
    return x * 2

print(slow(2), slow(2), len(calls))
t[0] = 11                                  # time passes
print(slow(2), len(calls))

Output

4 4 1
4 2

The clock is injected so the behaviour is testable without sleeping.

9. A context manager that restores state

from contextlib import contextmanager
import os

@contextmanager
def temp_env(**values):
    old = {k: os.environ.get(k) for k in values}
    os.environ.update({k: str(v) for k, v in values.items()})
    try:
        yield
    finally:
        for k, v in old.items():
            if v is None:
                os.environ.pop(k, None)
            else:
                os.environ[k] = v

with temp_env(MODE="test"):
    print(os.environ["MODE"])
print(os.environ.get("MODE"))

Output

test
None

10. Parse log lines and aggregate

import re
from collections import Counter, defaultdict
LOG = """\
2024-03-01 10:00:01 GET /home 200 12ms
2024-03-01 10:00:02 GET /cart 500 250ms
2024-03-01 10:00:03 POST /pay 200 80ms
2024-03-01 10:00:04 GET /cart 500 310ms
bad line
2024-03-01 10:00:05 GET /home 200 8ms
"""
pat = re.compile(r"^(?P<ts>\S+ \S+) (?P<method>[A-Z]+) (?P<path>\S+) (?P<status>\d{3}) (?P<ms>\d+)ms$")
status, slow = Counter(), defaultdict(list)
bad = 0
for line in LOG.splitlines():
    m = pat.match(line)
    if not m:
        bad += 1
        continue
    status[m["status"]] += 1
    slow[m["path"]].append(int(m["ms"]))
print(dict(status), "unparsed:", bad)
print({p: round(sum(v) / len(v), 1) for p, v in sorted(slow.items())})

Output

{'200': 3, '500': 2} unparsed: 1
{'/cart': 280.0, '/home': 10.0, '/pay': 80.0}

11. A bank account with validation (classes and exceptions)

from dataclasses import dataclass, field

class InsufficientFunds(Exception): pass

@dataclass
class Account:
    owner: str
    balance: int = 0
    history: list = field(default_factory=list)

    def deposit(self, amount):
        if amount <= 0:
            raise ValueError("amount must be positive")
        self.balance += amount
        self.history.append(("deposit", amount))

    def withdraw(self, amount):
        if amount > self.balance:
            raise InsufficientFunds(f"balance {self.balance}, requested {amount}")
        self.balance -= amount
        self.history.append(("withdraw", amount))

a = Account("Ada")
a.deposit(100); a.withdraw(30)
try:
    a.withdraw(500)
except InsufficientFunds as e:
    print("error:", e)
print(a.balance, a.history)

Output

error: balance 70, requested 500
70 [('deposit', 100), ('withdraw', 30)]

13. Validate and normalise phone numbers

import re
def normalise(raw, default_country="1"):
    digits = re.sub(r"\D", "", raw)
    if len(digits) == 10:
        digits = default_country + digits
    if not re.fullmatch(r"\d{11,15}", digits):
        raise ValueError(f"invalid phone: {raw!r}")
    return "+" + digits
for p in ["(555) 123-4567", "+44 20 7946 0958", "12-34"]:
    try:
        print(normalise(p))
    except ValueError as e:
        print(e)

14. A thread-safe counter

import threading
class Counter:
    def __init__(self):
        self._n, self._lock = 0, threading.Lock()
    def incr(self):
        with self._lock:
            self._n += 1
    @property
    def value(self): return self._n
c = Counter()
ts = [threading.Thread(target=lambda: [c.incr() for _ in range(5000)]) for _ in range(4)]
[t.start() for t in ts]; [t.join() for t in ts]
print(c.value)

Output

20000

15. Limit concurrent async tasks

import asyncio

async def main():
    sem = asyncio.Semaphore(3)
    running, peak = 0, 0
    async def task(i):
        nonlocal running, peak
        async with sem:
            running += 1
            peak = max(peak, running)
            await asyncio.sleep(0.01)
            running -= 1
        return i * i
    results = await asyncio.gather(*(task(i) for i in range(10)))
    print(results, "peak concurrency:", peak)
asyncio.run(main())

Output

[0, 1, 4, 9, 16, 25, 36, 49, 64, 81] peak concurrency: 3

16. From CSV text to a SQL report (Python and SQL together)

import csv, io, sqlite3
data = "region,product,units\nEU,pen,10\nEU,book,3\nUS,pen,7\nUS,pen,5\nUS,book,9\n"
con = sqlite3.connect(":memory:")
con.execute("CREATE TABLE sales (region TEXT, product TEXT, units INTEGER)")
rows = [(r["region"], r["product"], int(r["units"])) for r in csv.DictReader(io.StringIO(data))]
with con:
    con.executemany("INSERT INTO sales VALUES (?, ?, ?)", rows)
q = """
SELECT region, product, SUM(units) AS total,
       RANK() OVER (PARTITION BY region ORDER BY SUM(units) DESC) AS rnk
FROM sales GROUP BY region, product ORDER BY region, rnk
"""
for row in con.execute(q):
    print(row)

Output

('EU', 'pen', 10, 1)
('EU', 'book', 3, 2)
('US', 'pen', 12, 1)
('US', 'book', 9, 2)

17. Safe SQL search with optional filters

Build the WHERE clause from known fragments and pass values as parameters.

import sqlite3
def search(con, name=None, min_age=None, order="name"):
    sql, params = ["SELECT name, age FROM users WHERE 1=1"], []
    if name:
        sql.append("AND name LIKE ?"); params.append(f"%{name}%")
    if min_age is not None:
        sql.append("AND age >= ?"); params.append(min_age)
    col = {"name": "name", "age": "age"}[order]          # allow-listed identifier
    sql.append(f"ORDER BY {col}")
    return con.execute(" ".join(sql), params).fetchall()

con = sqlite3.connect(":memory:")
con.execute("CREATE TABLE users (name TEXT, age INTEGER)")
con.executemany("INSERT INTO users VALUES (?, ?)", [("ada", 36), ("ben", 25), ("alan", 41)])
print(search(con, name="a", min_age=30, order="age"))
print(search(con, name="x' OR '1'='1"))

Output

[('ada', 36), ('alan', 41)]
[]

18. Implement functools.reduce-style pipeline composition

from functools import reduce
def compose(*fns):
    return reduce(lambda f, g: lambda x: g(f(x)), fns)
pipeline = compose(str.strip, str.lower, lambda s: s.replace(" ", "-"))
print(pipeline("  Hello Big World  "))

Output

hello-big-world

How this gets asked

Which tool lazily merges several sorted iterables?

  • sorted(a + b + c) Builds everything in memory.
  • heapq.merge Correct.
  • zip Pairs items.
  • itertools.chain Concatenates, not merges.

heapq.merge returns an iterator that keeps the combined order.

How do you make a function like a TTL cache testable without sleeping?

  • Use real time.sleep Slow and flaky.
  • Inject the clock as a parameter Correct.
  • Disable the cache No.
  • Use threads No.

Dependency injection makes time controllable in tests.

Which is safe for an optional filter search in SQL from Python?

  • Format values into the string Injection risk.
  • Build the WHERE from fixed fragments and pass values as parameters Correct.
  • Escape quotes by hand Error-prone.
  • Use eval No.

Only fixed SQL fragments and bound values.

Interview questions: Python problems

8 questions. Try to answer aloud in under a minute, then reveal. Tags show the level. All questions are also in the interview bank.

Basic How would you count word frequencies?

Normalise with lower() and a regex, then collections.Counter and most_common or a sort by (-count, word).

Medium How do you group anagrams?

Use the sorted letters as a dictionary key with defaultdict(list).

Medium How do you flatten a nested list lazily?

A recursive generator with yield from for lists and tuples, yielding other items.

Medium How do you process a huge file?

Iterate over it line by line or in chunks with generators, keeping only aggregates in memory.

Medium Explain why this prints [2, 2, 2]: fs = [lambda: i for i in range(3)].

Late binding: each lambda looks up i when called, after the loop ended at 2. Bind with a default argument lambda i=i: i.

Advanced Implement a decorator with arguments.

A three-level function: the outer takes the arguments and returns a decorator that returns a wrapper, using functools.wraps.

Advanced How do you limit concurrent downloads in asyncio?

Wrap each task body in an asyncio.Semaphore and gather the tasks, or use a bounded queue with a fixed worker pool.

Advanced How do you write a safe dynamic SQL search with optional filters?

Append fixed SQL fragments for each provided filter, collect values in a parameter list, and allow-list any column used in ORDER BY.
